#!/usr/bin/env python3
"""Extract the CSR inventory snapshot from the AndeStar V5 SPA PDF.

Refresh tool for tests/andes_csr/csr_inventory.csv: parses Chapter 5's summary
tables (Tables 5-12, "Summary of Control & Status Registers") into one row per
CSR -- addr, privilege, name, mode, section. Run this when the spec revision
changes, then re-run classify_csr_coverage.py and the guard test.

Requires PyMuPDF (`pip install pymupdf`). Not used in CI -- the committed CSV is
the checked-in truth the guard test reads; this only regenerates it.

Usage:
  extract_csr_inventory.py <spec.pdf> [out.csv]   # default out: tests/andes_csr/csr_inventory.csv
"""
from __future__ import annotations

import csv
import re
import sys
from pathlib import Path

ADDR_RE = re.compile(r'^0x[0-9A-Fa-f]{3}(~0x[0-9A-Fa-f]{3}|~[0-9A-Fa-f]+)?$')
PRIV = {"MRW", "MRO", "MRW1C", "SRW", "SRO", "URW", "URO", "DRW", "DRO",
        "HRW", "HRO"}
TABLE_MODE = {
    "Table 5": "M-std", "Table 6": "M-andes", "Table 7": "D-std",
    "Table 8": "S-std", "Table 9": "S-andes", "Table 10": "U-std",
    "Table 11": "U-andes", "Table 12": "H-std",
}
# Bold group sub-headers that appear inside the summary tables.
SECTIONS = {
    "Machine Information Registers", "Machine Trap Setup", "Machine Trap Handling",
    "Machine Conﬁguration", "Machine Protection and Translation",
    "Machine Non-Maskable Interrupt Handling", "Counter Related CSRs",
    "Debug/Trace Registers", "CLIC Extension CSRs",
    "Machine Indirect CSR Access CSRs", "Machine State Management CSRs",
    "Conﬁguration Registers", "Crash Debug CSRs", "Memory CSRs",
    "Hardware Stack Protection & Recording", "Trap related CSR", "Control CSRs",
    "Enhanced CLIC CSRs", "Physical Memory Attribute CSRs",
    "Shadow Register CSRs", "Supervisor Trap Setup", "Supervisor Conﬁguration",
    "Supervisor Trap Handling", "Supervisor Protection and Translation",
    "State Management CSRs", "SPMP CSRs", "Supervisor Trap Related",
    "Supervisor Counter Related", "Supervisor Control",
    "Counter/Timer Registers", "Floating-Point CSRs", "ZCE CSRs",
    "N Extension CSRs", "Hypervisor Trap Setup",
    "Virtual supervisor Indirect CSR Access CSRs",
}
# Table 11 (AndeStar V5 User Mode) prints no sub-headers; relabel its rows.
ANDES_USER = {"uitb", "ucode", "uzobctl", "udcause", "ucctlbeginaddr",
              "ucctlcommand", "wfe", "sleepvalue", "txevt", "umisc_ctl"}


def extract(pdf_path):
    import fitz  # PyMuPDF
    doc = fitz.open(pdf_path)
    text = "".join(p.get_text() for p in doc)
    # The summary chapter: from Table 5 to the "Summary of CSR Existence" table.
    start = text.index("Table 5: RISC-V Standard Machine Mode CSRs")
    # "Summary of CSR Existence" also appears in the TOC (before start); take the
    # body occurrence that follows the Table 5 caption.
    end = text.index("Summary of CSR Existence", start)
    lines = [l.strip() for l in text[start:end].splitlines()]
    rows, mode, sec, i = [], "?", "?", 0
    while i < len(lines):
        l = lines[i]
        tm = [k for k in TABLE_MODE if l.startswith(k + ":")]
        if tm:
            mode = TABLE_MODE[tm[0]]; i += 1; continue
        if l in SECTIONS:
            sec = l; i += 1; continue
        if ADDR_RE.match(l) and i + 1 < len(lines):
            p = lines[i + 1]
            if p in PRIV or ("/" in p and p.split("/")[0] in PRIV):
                name = lines[i + 2] if i + 2 < len(lines) else "?"
                s = "Andes User Mode CSRs" if name in ANDES_USER else sec
                rows.append([l, p, name, mode, s]); i += 3; continue
        i += 1
    return rows


def main(argv):
    if not argv:
        print(__doc__)
        return 2
    pdf = argv[0]
    out = Path(argv[1]) if len(argv) > 1 else (
        Path(__file__).resolve().parent.parent
        / "tests/andes_csr/csr_inventory.csv")
    rows = extract(pdf)
    out.parent.mkdir(parents=True, exist_ok=True)
    with out.open("w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["addr", "priv", "name", "mode", "section"])
        w.writerows(rows)
    print(f"wrote {out} ({len(rows)} rows)")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

#!/usr/bin/env python3
"""Classify every spec CSR by its hybrid-handoff transport disposition.

The OFFLINE mirror of the wire-format transport set. It reads the spec inventory
snapshot (tests/andes_csr/csr_inventory.csv, produced by extract_csr_inventory.py)
and assigns each CSR a disposition under the supported envelope: bare-metal,
single-hart, M-mode or M+U-mode, trap-free ROI.

The policy here must match what include/hybrid/state_abi.h actually transports.
The guard test (tests/andes_csr/test_csr_coverage_complete.*) cross-checks the
two, so a CSR added to the ABI without a matching disposition (or vice versa)
fails CI -- the "nothing missed" flywheel.

Dispositions (see docs/csr_coverage.ipynb for the full legend):
  TRANSPORT_NOW  already carried in hybrid_state_v1
  NEW            writable M-mode plain state -- expansion target
  NEW_TRIG       debug-trigger CSR (shares the Debug Module; transported guarded)
  NEW_UDELTA     U-mode Andes control; transported, exercised only in M+U
  CONST_RO       read-only config/info -- engine derives it; nothing to exchange
  EXCL_RV32      RV32 high-half shadow -- irrelevant on RV64
  EXCL_SIDEFX    read/write has side effects -- not architectural state
  EXCL_W1C_HW    write-1-clear, hardware-driven (like mip)
  EXCL_INDIRECT  miselect/mireg access mechanism, not a direct register
  EXCL_NEXT      U-mode N-extension trap state (trap-free ROI + usually absent)
  EXCL_DEBUG     Debug-mode CSR the handoff itself drives
  EXCL_MODE      S-mode / Hypervisor -- outside the M / M+U envelope
  EXCL_TRAP      detailed trap-cause latch, live only after a trap

Usage:
  classify_csr_coverage.py <csr_inventory.csv>            # write csr_coverage.csv beside it
  classify_csr_coverage.py <csr_inventory.csv> --stdout   # print, do not write
"""
from __future__ import annotations

import csv
import sys
from pathlib import Path

# CSRs already carried in hybrid_state_v1 today (Bucket A). Names as the spec
# summary tables print them (range rows kept verbatim, e.g. pmpaddr1~63).

    "mstatus", "mepc", "mtvec", "mcause", "mtval", "mscratch", "medeleg",
    "mideleg", "mie", "mip", "mcycle", "minstret", "satp", "fcsr", "frm",
    "fﬂags",  # "fflags" with the fi-ligature the PDF emits
    "JVT", "mcache_ctl", "mmisc_ctl", "mxstatus", "mpft_ctl", "mclk_ctl",
    "milmb", "mdlmb", "mecc_code", "mhsp_ctl", "msp_bound", "msp_base",
    "mcountermask_m", "uitb", "ucode",
    "pmpcfg0", "pmpcfg1", "pmpcfg2", "pmpcfg3", "pmpcfg4~15",
    "pmpaddr0", "pmpaddr1~63",
    "pmacfg0", "pmacfg2", "pmaaddr0~47",

    "mintstatus": "EXCL_RO",

    "mscratchcswl": "EXCL_SIDEFX", "mcctlcommand": "EXCL_SIDEFX",
    "mintsel_jal": "EXCL_SIDEFX", "pushmcause": "EXCL_SIDEFX",
    "pushmepc": "EXCL_SIDEFX", "pushmxstatus": "EXCL_SIDEFX",
    "wfe": "EXCL_SIDEFX", "txevt": "EXCL_SIDEFX", "ucctlcommand": "EXCL_SIDEFX",
    "mcounterovf": "EXCL_W1C_HW",
    "miselect": "EXCL_INDIRECT", "mireg": "EXCL_INDIRECT",
    "mireg2": "EXCL_INDIRECT", "mireg3": "EXCL_INDIRECT",

    "mncause": "NEW", "mnstatus": "NEW", "mnvec": "NEW", "mrnxvec": "NEW",
    "mtvt": "NEW", "mintthresh": "NEW", "mcounteren": "NEW",
    "mcountinhibit": "NEW", "mhpmcounter3~6": "NEW", "mhpmevent3~6": "NEW",

    "uzobctl": "NEW_UDELTA", "umisc_ctl": "NEW_UDELTA",
    "sleepvalue": "NEW_UDELTA", "ucctlbeginaddr": "NEW_UDELTA",
    "udcause": "EXCL_TRAP",
}

RV32_ONLY = {
    "mcycleh", "minstreth", "mhpmcounter3h~6h", "mhpmevent3h~6h", "menvcfgh",
    "mseccfgh", "mstateen0h", "mstateen1h", "mstateen2h", "mstateen3h",
    "mcache_ctl2", "cycleh", "timeh", "instreth", "hpmcounter3h~6h",
    "pmacfg1", "pmacfg3", "pmacfg5", "pmacfg7", "pmacfg9", "pmacfg11",
}


def classify(addr, priv, name, mode, section):
    if name in TRANSPORT_NOW:
        return "TRANSPORT_NOW"
    if name in OVERRIDE:
        return OVERRIDE[name]
    if name in RV32_ONLY:
        return "EXCL_RV32"
    if mode in ("S-std", "S-andes", "H-std"):
        return "EXCL_MODE"
    if mode == "D-std" or name in ("dexc2dbg", "ddcause"):
        return "EXCL_DEBUG"
    if priv in ("MRO", "SRO", "URO", "DRO", "HRO"):
        return "CONST_RO"
    if mode == "U-std" and section == "N Extension CSRs":
        return "EXCL_NEXT"
    if mode == "U-std" and section == "Counter/Timer Registers":
        return "CONST_RO"
    raise SystemExit(f"unclassified CSR: {addr} {name} ({mode}/{section})")


def main(argv):
    if not argv:
        print(__doc__)
        return 2
    inv = Path(argv[0])
    to_stdout = "--stdout" in argv[1:]
    rows = list(csv.DictReader(inv.open()))
    out = [[r["addr"], r["priv"], r["name"], r["mode"], r["section"],
            classify(r["addr"], r["priv"], r["name"], r["mode"], r["section"])]
           for r in rows]
    if to_stdout:
        w = csv.writer(sys.stdout)
    else:
        dst = inv.with_name("csr_coverage.csv").open("w", newline="")
        w = csv.writer(dst)
    w.writerow(["addr", "priv", "name", "mode", "section", "decision"])
    w.writerows(out)
    if not to_stdout:
        print(f"wrote {inv.with_name('csr_coverage.csv')} ({len(out)} rows)")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

#!/usr/bin/env python3
"""Render docs/csr_coverage.ipynb from the coverage CSV, in spec-table order.

Walks tests/andes_csr/csr_coverage.csv in committed row order -- the AndeStar V5

NA_OVERRIDE = {
    "misa": "MRW but WARL-fixed to the engine's ISA; engine derives it, nothing to exchange",
    "tdata3": "MRW but hardwired to 0 (spec: always return 0)",
    "tinfo": "MRW but a read-only trigger-info register",
}


def status(name: str, dec: str):
    """(bucket, detail). bucket in {done, blocked, na}."""
    n = name.lower().replace("ﬂ", "fl").replace("ﬁ", "fi")
    if name in OUT_OF_ENVELOPE:
        return "na", OUT_OF_ENVELOPE[name] + " -> outside the M / M+U envelope"
    if n in ("fflags", "frm"):
        return "done", "rt_c_fcsr.c (packed in fcsr)"
    if name in DONE:
        return "done", DONE[name]
    if name in BLOCKED:
        return "blocked", "needs " + BLOCKED[name]
    if dec == "TRANSPORT_NOW":
        if name.startswith(("pmpcfg", "pmpaddr")):
            return "done", "handoff_roundtrip.S (M-mode PMP pass-through)"
        if name.startswith(("pmacfg", "pmaaddr")):
            return "done", "test_andes_sim_caps_parity.sh + handoff_roundtrip.S (Andes PMA block)"
        if name in ANDES_DONE:
            return "done", "test_andes_sim_caps_parity.sh + handoff_roundtrip.S (Andes block)"
        return "done", "handoff_roundtrip.S"
    if dec in ("NEW", "NEW_TRIG", "NEW_UDELTA"):
        return "blocked", "needs a capable engine"
    return "na", NA_OVERRIDE.get(name, NA[dec])


MARK = {"done": "[x] done", "blocked": "[~] blocked", "na": "[-] N/A"}

INTRO = """# Andes V5 CSR hybrid-handoff coverage

Review sheet for **every CSR in `AndeStar_V5_SPA_UM164_V1.7`**, in the PDF's
Chapter 5 summary-table order (Table 5 -> 12, sections top to bottom) so it reads
side by side with the spec. Envelope: bare-metal, single-hart, M-mode or
M+U-mode, trap-free ROI.

Auto-generated by `scripts/render_csr_matrix.py` from
`tests/andes_csr/csr_coverage.csv`; do not hand-edit. The guard
`tests/andes_csr/check_coverage.py` fails CI on any drift between the spec
inventory, the matrix data, and `include/hybrid/state_abi.h`.

## Status

Every CSR is exactly one of:

| status | meaning | the `detail` column shows |
|--------|---------|---------------------------|
| `[x] done` | the handoff handles it on the current engine (`ax45mpv_premium`) | **which test** proves it (or `handoff_roundtrip.S`); `(value not compared: ...)` when it is transported but, for a stated reason, not strict-compared |
| `[~] blocked` | we WANT to transport it, but this engine lacks the mode/extension | **what is missing** (e.g. `needs U-mode (e.g. d23)`, `needs CLIC (no built engine yet)`) |
| `[-] N/A` | we do NOT transport it, by design | **why** (read-only/WARL, RV32-only, side-effecting, S/H mode, ...) |

`priv` is the spec's read/write class. Some MRW CSRs are still `N/A` because the
implementation WARL-fixes or hardwires them (e.g. `misa` is pinned to the ISA),
so there is no state to exchange.
"""


def md_cell(src: str) -> dict:
    return {"cell_type": "markdown", "metadata": {}, "source": src.splitlines(keepends=True)}


def main() -> int:
    rows = list(csv.DictReader(COV.open()))
    cells = [md_cell(INTRO)]

    c = Counter(status(r["name"], r["decision"])[0] for r in rows)
    summ = ["## Counts\n\n", "| status | CSRs |\n", "|--------|------|\n"]
    for k in ("done", "blocked", "na"):
        summ.append(f"| {MARK[k]} | {c.get(k, 0)} |\n")
    summ.append(f"| **total** | **{len(rows)}** |\n")
    cells.append(md_cell("".join(summ)))

    cur_mode = cur_sec = None
    buf: list = []

    def flush():
        if buf:
            cells.append(md_cell("".join(buf)))

    for r in rows:
        if r["mode"] != cur_mode:
            flush(); buf.clear()
            cur_mode, cur_sec = r["mode"], None
            buf.append(f"## {TABLE.get(cur_mode, cur_mode)}\n")
        if r["section"] != cur_sec:
            cur_sec = r["section"]
            buf.append(f"\n### {cur_sec}\n\n")
            buf.append("| addr | name | priv | status | detail (test / missing / why) |\n")
            buf.append("|------|------|------|--------|-------------------------------|\n")
        st, detail = status(r["name"], r["decision"])
        buf.append(f"| `{r['addr']}` | {r['name']} | {r['priv']} | {MARK[st]} | {detail} |\n")
    flush()

    nb = {
        "cells": cells,
        "metadata": {
            "kernelspec": {"display_name": "Python 3", "language": "python",
                           "name": "python3"},
            "language_info": {"name": "python"},
        },
        "nbformat": 4, "nbformat_minor": 5,
    }
    OUT.write_text(json.dumps(nb, indent=1, ensure_ascii=False) + "\n")
    print(f"wrote {OUT} ({len(rows)} CSRs; "
          + ", ".join(f"{k}={c.get(k, 0)}" for k in ('done', 'blocked', 'na')) + ")")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

#!/usr/bin/env python3
"""Completeness guard for the Andes-V5 CSR transport coverage (the flywheel).

Cross-checks three artifacts and fails on any drift, so "nothing missed" is a
machine-checked invariant, not a hope:

  1. csr_inventory.csv  -- the spec snapshot (every CSR in the SPA Chapter 5)
  2. csr_coverage.csv   -- each CSR's transport disposition (the matrix)
  3. include/hybrid/state_abi.h -- what the wire format ACTUALLY transports

Checks (any failure exits non-zero):
  A. completeness  : matrix covers exactly the inventory's CSR set
  B. policy parity : re-running scripts/classify_csr_coverage.py reproduces the
                     committed matrix (offline policy == matrix)
  C. abi vs matrix : every CSR the ABI transports has a transport-disposition in
                     the matrix (never an excluded one), and every "carried
                     today" CSR is actually present in the ABI

NEW/NEW_TRIG/NEW_UDELTA CSRs not yet in the ABI are reported as PENDING (the
ROADMAP backlog), not failures -- the guard tightens automatically as each
expansion feature lands and the CSR appears in the ABI.

Usage: check_coverage.py [--abi PATH] [--quiet]
"""
from __future__ import annotations

import argparse
import csv
import re
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parent.parent
sys.path.insert(0, str(ROOT / "scripts"))
import classify_csr_coverage as policy  # noqa: E402

TRANSPORT = {"TRANSPORT_NOW", "NEW", "NEW_TRIG", "NEW_UDELTA"}

# Standard CSRs named as direct scalar fields in struct hybrid_state_v1.
STD_FIELDS = {
    "mstatus", "mepc", "mtvec", "mcause", "mtval", "mscratch", "medeleg",
    "mideleg", "mip", "mie", "mcycle", "minstret", "satp", "fcsr",
    "mcache_ctl", "mmisc_ctl",
}
# Array groups: matrix rows whose name starts with one of these are transported
# as a block (pmpcfg[]/pmpaddr[]/pmacfg[]/pmaaddr[] in the ABI).
ARRAY_PREFIX = ("pmpcfg", "pmpaddr", "pmacfg", "pmaaddr")
# V CSRs are transported but defined by the RVV spec, not this SPA -> not in the
# inventory; exclude them from the ABI<->matrix cross-check.
V_NAMES = {"vstart", "vtype", "vl", "vlenb", "vxsat", "vxrm", "vcsr", "vreg"}
# CSRs carried inside a composite ABI field, not as their own field. fcsr packs
# frm[7:5] | fflags[4:0], so both round-trip via the single fcsr field.
COMPOSITE = {"fflags": "fcsr", "frm": "fcsr"}


def norm(name: str) -> str:
    """Canonical CSR name across representations (case, fi/fl ligatures)."""
    n = name.strip().lower()
    return n.replace("ﬂ", "fl").replace("ﬁ", "fi")


# gdb/ABI name -> spec name, where the wire format uses a different mnemonic.
ABI_ALIAS = {"mcirq_entry": "mirq_entry"}


def matrix_keys(name: str) -> set:
    """All names a matrix row can be referenced by: each '/'-alternate, and each
    member of a 'prefixN~M' range (so the ABI's individual mhpmcounter3 matches
    the spec's mhpmcounter3~6 row)."""
    keys = set()
    for part in name.split("/"):
        part = part.strip()
        m = re.match(r"^(.*?)(\d+)~(\d+)$", part)
        if m:
            pre, a, b = m.group(1), int(m.group(2)), int(m.group(3))
            for i in range(a, b + 1):
                keys.add(norm(pre + str(i)))
        keys.add(norm(part))
    return keys


def abi_macro_csrs(abi_text: str) -> set:
    """Normalized names from every X(field, addr, "name") in a *_CSR_LIST macro."""
    out = set()
    for m in re.finditer(r'X\(\s*(\w+)\s*,\s*0x[0-9A-Fa-f]+\s*,\s*"[^"]*"\s*\)',
                         abi_text):
        out.add(norm(m.group(1)))
    return out


def load(path: Path) -> list:
    return list(csv.DictReader(path.open()))


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--abi", default=str(ROOT / "include/hybrid/state_abi.h"))
    ap.add_argument("--quiet", action="store_true")
    args = ap.parse_args()

    inv = load(HERE / "csr_inventory.csv")
    cov = load(HERE / "csr_coverage.csv")
    abi = Path(args.abi).read_text()
    errs: list = []

    # A. completeness: matrix covers exactly the spec inventory.
    inv_addrs = [r["addr"] for r in inv]
    if sorted(r["addr"] for r in cov) != sorted(inv_addrs):
        errs.append("A: coverage CSR set != inventory CSR set")
    inv_key = {(r["addr"], r["name"]) for r in inv}
    cov_key = {(r["addr"], r["name"]) for r in cov}
    if inv_key != cov_key:
        miss = inv_key - cov_key
        extra = cov_key - inv_key
        errs.append(f"A: inventory/coverage row mismatch missing={miss} extra={extra}")

    # B. policy parity: classify(inventory) reproduces the committed matrix.
    for ir, cr in zip(inv, cov):
        got = policy.classify(ir["addr"], ir["priv"], ir["name"], ir["mode"],
                              ir["section"])
        if got != cr["decision"]:
            errs.append(f"B: {ir['name']} classify={got} matrix={cr['decision']}")

    # C. abi vs matrix.
    abi_names = abi_macro_csrs(abi) | {norm(n) for n in STD_FIELDS}
    cov_by_name = {}
    for r in cov:
        for k in matrix_keys(r["name"]):
            cov_by_name[k] = r["decision"]
    # forward: nothing the ABI transports may be an excluded CSR
    for an in sorted(abi_names):
        if an in V_NAMES:
            continue
        dec = cov_by_name.get(ABI_ALIAS.get(an, an))
        if dec is None:
            errs.append(f"C: ABI transports '{an}' but it is not in the matrix")
        elif dec not in TRANSPORT:
            errs.append(f"C: ABI transports '{an}' but matrix says {dec} (excluded)")
    # reverse: every carried-today CSR must be present in the ABI
    abi_spec_names = {ABI_ALIAS.get(n, n) for n in abi_names}
    pending = []
    for r in cov:
        nn = norm(r["name"])
        pref = next((p for p in ARRAY_PREFIX if nn.startswith(p)), None)
        in_abi = bool(matrix_keys(r["name"]) & abi_spec_names) \
            or (pref is not None and f"{pref}[" in abi) \
            or (COMPOSITE.get(nn) in abi_names)
        if r["decision"] == "TRANSPORT_NOW":
            if nn in V_NAMES:
                continue
            if not in_abi:
                errs.append(f"C: carried-today CSR '{r['name']}' missing from ABI")
        elif r["decision"] in TRANSPORT and not in_abi:
            pending.append(r["name"])

    if errs:
        print("CSR COVERAGE GUARD: FAIL", file=sys.stderr)
        for e in errs:
            print("  - " + e, file=sys.stderr)
        return 1
    if not args.quiet:
        print(f"CSR coverage guard OK: {len(inv)} CSRs classified, "
              f"matrix==policy, ABI transports only matrix-approved CSRs")
        if pending:
            print(f"  PENDING (ROADMAP backlog, not yet in ABI): "
                  f"{len(pending)} -> {', '.join(pending)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env bash
# CSR coverage completeness guard (the "nothing missed" flywheel).
#
# Fails if the spec inventory snapshot, the coverage matrix, and the wire format
# (include/hybrid/state_abi.h) ever drift apart -- a CSR the spec adds, a CSR
# transported that the matrix marks excluded, or a carried-today CSR dropped from
# the ABI. See tests/andes_csr/check_coverage.py for the exact checks and
# docs/csr_coverage.ipynb for the disposition of every CSR.
set -uo pipefail

SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
ROOT=$(cd -- "$SCRIPT_DIR/../.." && pwd)
. "$SCRIPT_DIR/../lib/log.sh"

command -v python3 >/dev/null 2>&1 || fail "python3 required"

out=$(python3 "$SCRIPT_DIR/check_coverage.py" 2>&1) || fail "coverage guard FAILED"$'\n'"$out"
echo "$out"
ok "csr coverage guard: spec inventory == matrix == ABI (no drift)"
