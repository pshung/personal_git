// The pure core of `run --mode hybrid`: turn the CLI selection into a Trigger.
//
// Split out from cmd_run.cpp so it is unit-testable without spawning qemu/vsim.
// pc-mode boundaries are resolved against the ELF .symtab here (via elf_syms),
// so the live path in cmd_run only has to build the 2-step plan and run it.
#ifndef ANDESIM_HYBRID_HPP
#define ANDESIM_HYBRID_HPP

#include <string>
#include <vector>

#include "cli.hpp"
#include "engine_descriptor.hpp"
#include "plan.hpp"

namespace andes_sim {

// Build the ROI Trigger for a hybrid run from `cli` + `elf` (the ELF is needed
// to resolve pc-mode symbol boundaries). The pc boundaries are stored as
// "0x"-prefixed hex in params (enter_pc/exit_pc). On a missing/unresolvable pc
// boundary or unknown trigger, returns a Trigger with empty kind and sets `err`.
Trigger hybrid_trigger(const Cli& cli, const std::string& elf, std::string& err);

// Build the vsim accurate-leg's user-forwarded extra args (verbatim tokens
// appended after the trigger args): kanata ROI visualization
// (`--kanata <f> --kanata-roi [--objdump <f>]`). `kind` is the trigger kind,
// used to validate: --kanata is marker/pc only; --objdump requires --kanata.
// On a violation returns {} and sets `err`. Absent kanata yields an empty
// list (the common case). Post-`--` args are NOT forwarded here -- hybrid
// rejects them up front (cmd_run), as the handoff owns both legs' argv.
std::vector<std::string> vsim_extra_args(const Cli& cli, const std::string& kind,
                                         std::string& err);

// ISA-sync guard for the accurate-leg engine: the hybrid fast leg is a 64-bit
// QEMU, so an engine reporting xlen != 64 cannot share architectural state
// with it. Returns "" when the pairing is valid, else the diagnostic. An
// engine that reports no xlen (0) is accepted as-is.
std::string hybrid_hw_check(const EngineDescriptor& acc);

}  // namespace andes_sim

#endif  // ANDESIM_HYBRID_HPP
===== driver/hybrid.cpp =====
// See hybrid.hpp. Builds the ROI Trigger from the CLI, resolving pc boundaries.
#include "hybrid.hpp"

#include <cstdint>

#include "elf_syms.hpp"

namespace andes_sim {
namespace {

// Numeric value of a bare-hex PC string from resolve_pc. resolve_pc already
// validated the digits; an absurd >64-bit address (out_of_range) yields false,
// so the equal-ROI guard simply skips it (the bogus address fails downstream).
bool hex_pc(const std::string& s, uint64_t& out) {
  try {
    std::size_t pos = 0;
    out = std::stoull(s, &pos, 16);
    return pos == s.size();
  } catch (...) {
    return false;
  }
}

}  // namespace

Trigger hybrid_trigger(const Cli& cli, const std::string& elf,
                       std::string& err) {
  err.clear();
  // --trigger is REQUIRED for hybrid: no silent default. An absent trigger used
  // to fall back to marker, which masked running an unmarked ELF (the fast leg
  // ran to completion instead of draining at the ROI, surfacing only as a
  // cryptic "expected exit 200, got 0"). Make the user name the ROI delimiter.
  if (cli.trigger.empty()) {
    err = "--trigger marker|pc is required "
          "(the ROI delimiter; see 'andesim run --help')";
    return {};
  }
  const std::string kind = cli.trigger;

  if (kind == "marker") return {"marker", {}};  // markers in the workload; no params

  if (kind == "pc") {
    if (cli.pc_start.empty() || cli.pc_end.empty()) {
      err = "trigger pc requires --pc-start and --pc-end "
            "(a symbol name or 0x address)";
      return {};
    }
    std::string e1, e2;
    const std::string enter = resolve_pc(elf, cli.pc_start, e1);
    if (!e1.empty()) { err = e1; return {}; }
    const std::string exit = resolve_pc(elf, cli.pc_end, e2);
    if (!e2.empty()) { err = e2; return {}; }
    // A zero-length ROI (enter == exit) is degenerate: vsim resumes at the
    // enter PC and the exit trigger fires on that very instruction, so the
    // window measures nothing. Compare numerically -- a symbol and its literal
    // 0x address (or 0x0001000 vs 0x1000) are different strings, same PC.
    uint64_t a_enter = 0, a_exit = 0;
    if (hex_pc(enter, a_enter) && hex_pc(exit, a_exit) && a_enter == a_exit) {
      err = "trigger pc: --pc-start and --pc-end resolve to the same address "
            "0x" + enter + " (an empty ROI measures nothing)";
      return {};
    }
    // Store with the 0x prefix the plugin (enter_pc=0x..) and vsim
    // (--hybrid-exit-pc 0x..) expect; leg_trigger_args concatenates verbatim.
    return {"pc", {{"enter_pc", "0x" + enter}, {"exit_pc", "0x" + exit}}};
  }

  err = "unknown trigger '" + kind + "' (want marker|pc)";
  return {};
}

std::string hybrid_hw_check(const EngineDescriptor& acc) {
  if (acc.xlen != 0 && acc.xlen != 64)
    return "engine " + acc.id + " is rv" + std::to_string(acc.xlen) +
           " (xlen=" + std::to_string(acc.xlen) +
           "); the hybrid fast leg (QEMU) is rv64-only";
  return "";
}

std::vector<std::string> vsim_extra_args(const Cli& cli, const std::string& kind,
                                         std::string& err) {
  err.clear();
  std::vector<std::string> out;

  // kanata ROI visualization first (marker/pc only; objdump rides kanata).
  if (!cli.kanata.empty()) {
    if (kind != "marker" && kind != "pc") {
      err = "--kanata is only supported with trigger marker or pc (got " + kind +
            ")";
      return {};
    }
    out.push_back("--kanata");
    out.push_back(cli.kanata);
    out.push_back("--kanata-roi");
    if (!cli.objdump.empty()) {
      out.push_back("--objdump");
      out.push_back(cli.objdump);
    }
  } else if (!cli.objdump.empty()) {
    err = "--objdump requires --kanata";
    return {};
  }

  return out;
}

}  // namespace andes_sim
===== driver/elf_syms.hpp =====
// ELF symbol resolution for pc-trigger boundaries (F9).
//
// pc mode switches QEMU <-> vsim at two PCs. Either boundary may be a
// "0x"-prefixed hex address (used verbatim) OR a symbol defined in the ELF
// (resolved against its .symtab). This decouples the switch points from the
// workload source -- any function can bound the ROI, with no planted markers.
//
// C++ port of tests/lib/roi.sh:resolve_pc, reading the ELF .symtab directly via
// <elf.h> rather than shelling out to `nm`.
#ifndef ANDESIM_ELF_SYMS_HPP
#define ANDESIM_ELF_SYMS_HPP

#include <cstdint>
#include <string>

namespace andes_sim {

// Resolve a pc-mode boundary to a bare hex address string (no "0x" prefix).
//   "0x<h>" / "0X<h>"  -> "<h>" verbatim (prefix stripped, digits unchanged)
//   <symbol>           -> the symbol's st_value as lowercase hex
// On any failure (undefined symbol, unreadable/!ELF file) returns "" and sets
// `err` to a message naming the boundary; on success leaves `err` empty.
std::string resolve_pc(const std::string& elf_path, const std::string& boundary,
                       std::string& err);

// The ELF entry point (e_entry) -- the default lock-step arm pc. Returns 0 and
// sets `err` on an unreadable / non-RV64 file; clears `err` on success.
uint64_t elf_entry(const std::string& elf_path, std::string& err);

// The ELF class width in bits from e_ident[EI_CLASS]: 32 (ELFCLASS32) or 64
// (ELFCLASS64). Returns 0 when `elf_path` is unreadable or not an ELF. Reads
// only the 16-byte e_ident (the class-independent prefix), so it answers for
// rv32 and rv64 alike -- the source of truth for the fast auto-pick of
// qemu-system-riscv32/64 and the cycle/hybrid arch-match guard.
int elf_xlen(const std::string& elf_path);

// L26/F3: the highest byte of the ELF's static memory image -- the maximum
// (p_paddr + p_memsz) over PT_LOAD segments. The driver rejects a hybrid run
// when this exceeds the V-scratch GPA (ram_base + --mem-size): the static image
// would extend into the reserved transport page above the program's RAM.
// Returns 0 + err on an unreadable / non-ELFCLASS64 file; 0 with empty err when
// the ELF has no loadable segments (nothing to clash with the scratch).
uint64_t elf_load_top(const std::string& elf_path, std::string& err);

// Reverse lookup: the nearest function symbol at or below `addr`, formatted
// "name+0xoff" (or just "name" at offset 0). Returns "" when no covering
// function symbol is found or the ELF is unreadable -- the caller always has
// the raw pc to fall back on, so this never errors.
std::string symbolize(const std::string& elf_path, uint64_t addr);

// Returns {st_value, st_value + st_size} for the FUNC symbol covering `addr`.
// Both members are 0 when not found or unreadable.
std::pair<uint64_t, uint64_t> fn_range(const std::string& elf_path, uint64_t addr);

}  // namespace andes_sim

#endif  // ANDESIM_ELF_SYMS_HPP
===== driver/elf_syms.cpp =====
// See elf_syms.hpp. Reads the ELF .symtab directly (no `nm` shell-out).
#include "elf_syms.hpp"

#include <cstdint>
#include <cstring>
#include <fstream>
#include <sstream>
#include <vector>

#include <elf.h>

namespace andes_sim {
namespace {

// Lowercase, minimal-width hex -- matches the bare hex a stripped "0x" address
// yields, so a symbol and its literal address resolve to the same string.
std::string to_hex(uint64_t v) {
  std::ostringstream ss;
  ss << std::hex << v;
  return ss.str();
}

bool slurp(const std::string& path, std::vector<char>& buf) {
  std::ifstream f(path, std::ios::binary);
  if (!f) return false;
  std::ostringstream ss;
  ss << f.rdbuf();
  const std::string s = ss.str();
  buf.assign(s.begin(), s.end());
  return true;
}

// memcpy a POD header out of the buffer at `off`; false if it would overrun
// (buffers may be truncated/malformed -- never read past the end).
template <class T>
bool load(const std::vector<char>& buf, size_t off, T& out) {
  if (off + sizeof(T) > buf.size()) return false;
  std::memcpy(&out, buf.data() + off, sizeof(T));
  return true;
}

// Slurp `path` and validate its ELFCLASS64 header into `eh`. Returns false and
// sets `err` (unreadable / not-ELF / not-RV64) on failure, clears `err` on ok.
// Shared prologue for every ELF reader below.
bool read_elf64_header(const std::string& path, std::vector<char>& buf,
                       Elf64_Ehdr& eh, std::string& err) {
  err.clear();
  if (!slurp(path, buf)) {
    err = "cannot read ELF '" + path + "'";
    return false;
  }
  if (!load(buf, 0, eh) || std::memcmp(eh.e_ident, ELFMAG, SELFMAG) != 0) {
    err = "not an ELF file: " + path;
    return false;
  }
  if (eh.e_ident[EI_CLASS] != ELFCLASS64) {
    err = "not an RV64 (ELFCLASS64) file: " + path;
    return false;
  }
  return true;
}

}  // namespace

std::string resolve_pc(const std::string& elf_path, const std::string& boundary,
                       std::string& err) {
  err.clear();

  // A "0x"/"0X" address is used verbatim with the prefix stripped (roi.sh
  // `${boundary#0[xX]}`); the .symtab is never touched.
  if (boundary.starts_with("0x") || boundary.starts_with("0X")) {
    std::string addr = boundary.substr(2);
    // Must be all hex digits: a verbatim pass of e.g. "ZZZ"/"10zz" would let the
    // downstream strtoull silently truncate to 0x0/0x10 (wrong PC window).
    if (addr.empty() ||
        addr.find_first_not_of("0123456789abcdefABCDEF") != std::string::npos) {
      err = "malformed address '" + boundary + "'";
      return "";
    }
    return addr;
  }

  std::vector<char> buf;
  Elf64_Ehdr eh;
  if (!read_elf64_header(elf_path, buf, eh, err)) return "";

  // Scan section headers for a symbol table; resolve names through its linked
  // string table (sh_link).
  for (unsigned i = 0; i < eh.e_shnum; ++i) {
    Elf64_Shdr sym_sh;
    if (!load(buf, eh.e_shoff + static_cast<size_t>(i) * eh.e_shentsize, sym_sh))
      break;
    if (sym_sh.sh_type != SHT_SYMTAB || sym_sh.sh_entsize == 0) continue;

    Elf64_Shdr str_sh;
    if (sym_sh.sh_link >= eh.e_shnum ||
        !load(buf, eh.e_shoff + static_cast<size_t>(sym_sh.sh_link) * eh.e_shentsize,
              str_sh))
      continue;

    const size_t nsym = sym_sh.sh_size / sym_sh.sh_entsize;
    for (size_t s = 0; s < nsym; ++s) {
      Elf64_Sym sym;
      if (!load(buf, sym_sh.sh_offset + s * sym_sh.sh_entsize, sym)) break;
      if (sym.st_shndx == SHN_UNDEF) continue;  // --defined-only

      const size_t noff = str_sh.sh_offset + sym.st_name;
      if (noff >= buf.size()) continue;
      const char* name = buf.data() + noff;
      const size_t avail = buf.size() - noff;
      if (::strnlen(name, avail) == avail) continue;  // unterminated in bounds
      if (boundary == name) return to_hex(sym.st_value);
    }
  }

  err = "symbol '" + boundary + "' not defined in " + elf_path;
  return "";
}

uint64_t elf_entry(const std::string& elf_path, std::string& err) {
  std::vector<char> buf;
  Elf64_Ehdr eh;
  if (!read_elf64_header(elf_path, buf, eh, err)) return 0;
  return eh.e_entry;
}

int elf_xlen(const std::string& elf_path) {
  // Only e_ident is needed (16 bytes, identical layout for both classes), so
  // read just that rather than slurping a whole (possibly large) ELF.
  std::ifstream f(elf_path, std::ios::binary);
  unsigned char ident[EI_NIDENT];
  if (!f.read(reinterpret_cast<char*>(ident), EI_NIDENT)) return 0;
  if (std::memcmp(ident, ELFMAG, SELFMAG) != 0) return 0;  // not an ELF
  switch (ident[EI_CLASS]) {
    case ELFCLASS32: return 32;
    case ELFCLASS64: return 64;
    default: return 0;
  }
}

uint64_t elf_load_top(const std::string& elf_path, std::string& err) {
  std::vector<char> buf;
  Elf64_Ehdr eh;
  if (!read_elf64_header(elf_path, buf, eh, err)) return 0;
  // Top of the static image: the highest (p_paddr + p_memsz) over PT_LOAD.
  // p_paddr is the guest-physical placement (== p_vaddr for the identity-mapped
  // bare-metal fixtures), the space the V-scratch GPA also lives in.
  uint64_t top = 0;
  for (uint16_t i = 0; i < eh.e_phnum; ++i) {
    Elf64_Phdr ph;
    const size_t off = static_cast<size_t>(eh.e_phoff) +
                       static_cast<size_t>(i) * eh.e_phentsize;
    if (!load(buf, off, ph)) {
      err = "truncated program header table in '" + elf_path + "'";
      return 0;
    }
    if (ph.p_type != PT_LOAD) continue;
    const uint64_t end = ph.p_paddr + ph.p_memsz;
    if (end > top) top = end;
  }
  return top;  // 0 (empty err) when the ELF has no loadable segments
}

std::string symbolize(const std::string& elf_path, uint64_t addr) {
  std::vector<char> buf;
  Elf64_Ehdr eh;
  std::string ignored;  // symbolize never errors -- empty result is the signal
  if (!read_elf64_header(elf_path, buf, eh, ignored)) return "";

  // Track the highest-valued FUNC symbol whose range covers addr (st_value <=
  // addr < st_value + st_size), falling back to the nearest preceding FUNC
  // when sizes are absent.
  std::string best_name;
  uint64_t best_value = 0;
  bool found = false;

  for (unsigned i = 0; i < eh.e_shnum; ++i) {
    Elf64_Shdr sym_sh;
    if (!load(buf, eh.e_shoff + static_cast<size_t>(i) * eh.e_shentsize, sym_sh))
      break;
    if (sym_sh.sh_type != SHT_SYMTAB || sym_sh.sh_entsize == 0) continue;

    Elf64_Shdr str_sh;
    if (sym_sh.sh_link >= eh.e_shnum ||
        !load(buf, eh.e_shoff + static_cast<size_t>(sym_sh.sh_link) * eh.e_shentsize,
              str_sh))
      continue;

    const size_t nsym = sym_sh.sh_size / sym_sh.sh_entsize;
    for (size_t s = 0; s < nsym; ++s) {
      Elf64_Sym sym;
      if (!load(buf, sym_sh.sh_offset + s * sym_sh.sh_entsize, sym)) break;
      if (sym.st_shndx == SHN_UNDEF) continue;
      if (ELF64_ST_TYPE(sym.st_info) != STT_FUNC) continue;
      if (sym.st_value > addr) continue;
      // With a size, require coverage; without, accept as a preceding label.
      if (sym.st_size != 0 && addr >= sym.st_value + sym.st_size) continue;
      if (found && sym.st_value <= best_value) continue;

      const size_t noff = str_sh.sh_offset + sym.st_name;
      if (noff >= buf.size()) continue;
      const char* name = buf.data() + noff;
      const size_t avail = buf.size() - noff;
      if (::strnlen(name, avail) == avail) continue;

      best_name = name;
      best_value = sym.st_value;
      found = true;
    }
  }

  if (!found) return "";
  const uint64_t off = addr - best_value;
  if (off == 0) return best_name;
  std::ostringstream ss;
  ss << best_name << "+0x" << std::hex << off;
  return ss.str();
}

std::pair<uint64_t, uint64_t> fn_range(const std::string& elf_path, uint64_t addr) {
  std::vector<char> buf;
  Elf64_Ehdr eh;
  std::string ignored;
  if (!read_elf64_header(elf_path, buf, eh, ignored)) return {0, 0};

  uint64_t best_value = 0;
  uint64_t best_size = 0;
  bool found = false;

  for (unsigned i = 0; i < eh.e_shnum; ++i) {
    Elf64_Shdr sym_sh;
    if (!load(buf, eh.e_shoff + static_cast<size_t>(i) * eh.e_shentsize, sym_sh)) break;
    if (sym_sh.sh_type != SHT_SYMTAB || sym_sh.sh_entsize == 0) continue;
    const size_t nsym = sym_sh.sh_size / sym_sh.sh_entsize;
    for (size_t s = 0; s < nsym; ++s) {
      Elf64_Sym sym;
      if (!load(buf, sym_sh.sh_offset + s * sym_sh.sh_entsize, sym)) break;
      if (sym.st_shndx == SHN_UNDEF) continue;
      if (ELF64_ST_TYPE(sym.st_info) != STT_FUNC) continue;
      if (sym.st_value > addr) continue;
      if (sym.st_size != 0 && addr >= sym.st_value + sym.st_size) continue;
      if (found && sym.st_value <= best_value) continue;
      best_value = sym.st_value;
      best_size = sym.st_size;
      found = true;
    }
  }

  if (!found || best_size == 0) return {0, 0};
  return {best_value, best_value + best_size};
}

}  // namespace andes_sim

// Reader for a drained hybrid_state_v1 file -- the C++ analog of
// orchestrator.abi.validate_state_file + hsim._read_state_mcycle. Header-only:
// it #includes the shared ABI directly, so mcycle/minstret/flags/vlenb are
// struct members with no hand-copied offsets.
#ifndef ANDESIM_STATE_HPP
#define ANDESIM_STATE_HPP

#include <cstdio>
#include <vector>
#include <string>

#include "hybrid/state_abi.h"

namespace andes_sim {

struct StateFile {
  hybrid_state_v1 st{};
  std::string error;  // empty => valid; non-empty describes the rejection
  bool ok() const { return error.empty(); }
};

// Read `path` and validate it (in order): byte size == sizeof(hybrid_state_v1),
// magic == HYBRID_STATE_MAGIC, version == HYBRID_STATE_VERSION. On any failure
// `error` is set and `st` is unspecified; otherwise `st` holds the full state.
inline StateFile read_state_file(const std::string& path) {
  StateFile r;
  std::FILE* f = std::fopen(path.c_str(), "rb");
  if (!f) {
    r.error = path + ": cannot open";
    return r;
  }
  std::fseek(f, 0, SEEK_END);
  const long sz = std::ftell(f);
  std::rewind(f);
  if (sz != static_cast<long>(sizeof(hybrid_state_v1))) {
    r.error = path + ": size " + std::to_string(sz) + " != expected " +
              std::to_string(sizeof(hybrid_state_v1));
    std::fclose(f);
    return r;
  }
  const size_t n = std::fread(&r.st, 1, sizeof(r.st), f);
  std::fclose(f);
  if (n != sizeof(r.st)) {
    r.error = path + ": short read";
    return r;
  }

  char buf[256];
  if (r.st.magic != HYBRID_STATE_MAGIC) {
    std::snprintf(buf, sizeof(buf), "%s: magic 0x%08x != expected 0x%08x",
                  path.c_str(), r.st.magic, HYBRID_STATE_MAGIC);
    r.error = buf;
    return r;
  }
  if (r.st.version != HYBRID_STATE_VERSION) {
    std::snprintf(buf, sizeof(buf), "%s: version %u != expected %u",
                  path.c_str(), r.st.version, HYBRID_STATE_VERSION);
    r.error = buf;
    return r;
  }
  return r;
}

}  // namespace andes_sim

#endif  // ANDESIM_STATE_HPP
===== driver/shm.hpp =====
// Shared-mmap allocation for andesim.
//
// The caller asks for `size` bytes; we resolve a backing directory over a
// tier list (first fit wins, fit = size + 256 MiB free), allocate a sparse
// file with mkstemp+truncate, and print one status line that always names
// the chosen tier -- and, when we fell past /dev/shm, why. HSIM_MMAP_DIR
// pins a single tier (hard error if it does not fit; no fallback past it).
#ifndef ANDESIM_SHM_HPP
#define ANDESIM_SHM_HPP

#include <functional>
#include <string>
#include <vector>

namespace andes_sim {

// Free-space headroom a tier must have beyond the requested size.
inline constexpr long long kMmapSafetyMargin = 256LL * 1024 * 1024;

// One backing-store candidate: the directory plus the label used in
// status/error messages ("/dev/shm", "~/.cache/andesim", "HSIM_MMAP_DIR=..").
struct MmapTier {
  std::string dir;
  std::string label;
};

// resolve_mmap_dir result: the chosen dir ("" on failure) and, when a tier
// was skipped, a human note saying which and why ("" when tiers[0] fit).
struct MmapDirChoice {
  std::string dir;
  std::string note;
};

// Parse "8G" / "512M" / "1024" / "1.5G" into a positive byte count. On success
// returns the count and leaves `err` empty; on failure returns 0 and sets `err`
// to the argparse-style message. Mirrors hsim_shm.parse_size (fractional values
// truncate, like Python int(float(num) * mult)).
long long parse_size(const std::string& spec, std::string& err);

// L26/F2: the physical shared mmap is one V-scratch page (HYBRID_V_SCRATCH_BYTES)
// larger than the program's --mem-size, so the V-register transport page sits
// just ABOVE the program's declared RAM. No program layout -- low stack, top-of-
// RAM stack, or a heap grown to the ceiling -- can reach a page past its own
// RAM, so the scratch can never be clobbered. The vsim side parks the scratch
// at the mmap's top page (v_scratch_gpa), which equals ram_base + program_bytes.
long long physical_mmap_bytes(long long program_bytes);

// The largest --mem-size whose physical mmap still fits the 2 GiB AE350 DRAM
// window (program_bytes + scratch page <= 2 GiB). --mem-size is capped here.
long long max_program_mem_bytes();

// First-fit resolver over `tiers`. `probe(dir)` returns the dir's free bytes,
// or -1 when it does not exist. A tier fits when probe >= size + margin.
// On success `err` is empty; when nothing fits, returns {"", ""} and `err`
// names every tier's reason plus the HSIM_MMAP_DIR hint. Pure: no env, no
// filesystem -- testable with a fake probe.
MmapDirChoice resolve_mmap_dir(
    const std::vector<MmapTier>& tiers, long long size,
    const std::function<long long(const std::string&)>& probe,
    std::string& err);

// Build the real tier list from the environment:
//   HSIM_MMAP_DIR set -> that single pinned tier;

// Free bytes on the filesystem holding `dir`, or -1 if it does not exist.
long long free_bytes(const std::string& dir) {
  struct statvfs vfs;
  if (statvfs(dir.c_str(), &vfs) != 0) return -1;
  return static_cast<long long>(vfs.f_bavail) *
         static_cast<long long>(vfs.f_frsize);
}

// Human-readable byte count: "8.0 GiB", "512.0 MiB", "1024 B". Mirrors
// hsim_shm._format_bytes (one decimal, TiB/GiB/MiB/KiB, else "N B").
std::string format_bytes(long long n) {
  struct Unit {
    const char* suffix;
    long long mul;
  };
  static const Unit units[] = {
      {"TiB", 1024LL * 1024 * 1024 * 1024},
      {"GiB", 1024LL * 1024 * 1024},
      {"MiB", 1024LL * 1024},
      {"KiB", 1024LL},
  };
  for (const Unit& u : units) {
    if (n >= u.mul) {
      char buf[64];
      std::snprintf(buf, sizeof(buf), "%.1f %s",
                    static_cast<double>(n) / static_cast<double>(u.mul),
                    u.suffix);
      return buf;
    }
  }
  return std::to_string(n) + " B";
}

}  // namespace

long long parse_size(const std::string& spec, std::string& err) {
  static const std::regex re(R"(\s*(\d+(?:\.\d+)?)\s*([KMGT]?)\s*)",
                             std::regex::icase);
  std::smatch m;
  if (!std::regex_match(spec, m, re)) {
    err = "invalid mmap size '" + spec +
          "': expected <number>[K|M|G|T] (e.g. '128M', '8G', '1024')";
    return 0;
  }
  const double num = std::stod(m[1].str());
  long long mult = 1;
  if (m[2].length()) {
    switch (std::toupper(static_cast<unsigned char>(m[2].str()[0]))) {
      case 'K': mult = 1024LL; break;
      case 'M': mult = 1024LL * 1024; break;
      case 'G': mult = 1024LL * 1024 * 1024; break;
      case 'T': mult = 1024LL * 1024 * 1024 * 1024; break;
    }
  }
  // Truncate toward zero like Python int(float(num) * multiplier).
  const long long size =
      static_cast<long long>(num * static_cast<double>(mult));
  if (size <= 0) {
    err = "invalid mmap size '" + spec + "': must be > 0";
    return 0;
  }
  err.clear();
  return size;
}

// L26/F2: physical mmap = program RAM + one V-scratch page. See shm.hpp.
long long physical_mmap_bytes(long long program_bytes) {
  return program_bytes + HYBRID_V_SCRATCH_BYTES;
}

long long max_program_mem_bytes() {
  // AE350 DRAM window is 2 GiB; leave room for the scratch page on top.
  constexpr long long kAe350DramWindowBytes = 2LL * 1024 * 1024 * 1024;
  return kAe350DramWindowBytes - HYBRID_V_SCRATCH_BYTES;
}

std::vector<std::string> wsl_warnings(const std::string& proc_version,
                                      const std::string& repo_dir,
                                      const std::vector<MmapTier>& tiers) {
  std::string lower = proc_version;
  for (char& c : lower) c = static_cast<char>(std::tolower(
      static_cast<unsigned char>(c)));
  if (lower.find("microsoft") == std::string::npos) return {};

  std::vector<std::string> warns;
  warns.push_back(
      "WSL2: /dev/shm is capped by the WSL VM memory; raise memory= in "
      ".wslconfig if tier 1 is too small");
  if (repo_dir.starts_with("/mnt/"))
    warns.push_back("repo is under /mnt/ (drvfs): slow, and MAP_SHARED is "
                    "not coherent -- keep the repo in the Linux home (~)");
  for (const MmapTier& t : tiers)
    if (t.dir.starts_with("/mnt/"))
      warns.push_back("mmap dir " + t.dir +
                      " is under /mnt/ (drvfs): MAP_SHARED is not coherent "
                      "there -- use a Linux-side dir");
  return warns;
}

std::vector<std::string> qemu_m_args(const std::string& spec,
                                     std::string& err) {
  err.clear();
  if (spec.empty()) return {};
  const long long bytes = parse_size(spec, err);
  if (!err.empty()) {
    err = "--mem-size: " + err;
    return {};
  }
  if (bytes % (1024 * 1024) != 0) {
    err = "--mem-size '" + spec +
          "': QEMU's -m takes whole MiB; round to a multiple of 1M";
    return {};
  }
  return {"-m", std::to_string(bytes / (1024 * 1024)) + "M"};
}

MmapDirChoice resolve_mmap_dir(
    const std::vector<MmapTier>& tiers, long long size,
    const std::function<long long(const std::string&)>& probe,
    std::string& err) {
  const long long need = size + kMmapSafetyMargin;
  std::vector<std::string> reasons;
  for (const MmapTier& t : tiers) {
    const long long free = probe(t.dir);
    if (free >= need) {
      err.clear();
      std::string note;
      for (size_t i = 0; i < reasons.size(); i++)
        note += (i ? "; " : "") + reasons[i];
      return {t.dir, note};
    }
    reasons.push_back(free < 0 ? t.label + " missing"
                               : t.label + " too small (" +
                                     format_bytes(free) + " free)");
  }
  err = "no mmap directory can hold " + format_bytes(size) +
        " + 256 MiB margin: ";
  for (size_t i = 0; i < reasons.size(); i++)
    err += (i ? "; " : "") + reasons[i];
  err += "; set HSIM_MMAP_DIR to a directory with enough free space";
  return {"", ""};
}

std::vector<MmapTier> mmap_tiers() {
  if (const char* pinned = std::getenv("HSIM_MMAP_DIR"); pinned && pinned[0])
    return {{pinned, "HSIM_MMAP_DIR='" + std::string(pinned) + "'"}};

  std::string cache;
  if (const char* xdg = std::getenv("XDG_CACHE_HOME"); xdg && xdg[0])
    cache = std::string(xdg) + "/andesim";
  else if (const char* home = std::getenv("HOME"); home && home[0])
    cache = std::string(home) + "/.cache/andesim";

  std::vector<MmapTier> tiers = {{"/dev/shm", "/dev/shm"}};
  if (!cache.empty()) {
    std::error_code ec;
    std::filesystem::create_directories(cache, ec);  // probe needs it to exist
    tiers.push_back({cache, cache});
  }
  return tiers;
}

std::string allocate(long long size, std::string& err,
                     const std::string& prefix) {
  const MmapDirChoice choice =
      resolve_mmap_dir(mmap_tiers(), size, free_bytes, err);
  if (choice.dir.empty()) return "";

  std::string tmpl = choice.dir + "/" + prefix + "XXXXXX";
  std::vector<char> buf(tmpl.begin(), tmpl.end());
  buf.push_back('\0');
  const int fd = mkstemp(buf.data());
  if (fd < 0) {
    err = "failed to create mmap file in " + choice.dir;
    return "";
  }
  close(fd);
  const std::string path(buf.data());
  if (truncate(path.c_str(), size) != 0) {
    unlink(path.c_str());
    err = "failed to size mmap file " + path;
    return "";
  }

  std::string line = "[hsim mmap] " + format_bytes(size) + " at " +
                     choice.dir + " (free: " + format_bytes(free_bytes(choice.dir)) + ")";
  if (!choice.note.empty()) line += " (fallback: " + choice.note + ")";
  std::printf("%s\n", line.c_str());
  std::fflush(stdout);
  err.clear();
  return path;
}
