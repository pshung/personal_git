+      CHECK(r.diffs[0].field == "pmacfg1");
+    }
+    {
+      hybrid_state_v1 got = ref;
+      got.andes.pmaaddr[5] = 0xbad;
+      const DiffReport r = diff_states(ref, got);
+      CHECK(r.diffs.size() == 1);
+      CHECK(r.diffs[0].field == "pmaaddr5");
+    }
+  }
+
+  {  // without HYBRID_FLAG_ANDES the block is NOT compared (zero or stale)
+    hybrid_state_v1 ref = andes_state();
+    ref.flags = HYBRID_FLAG_RV64;  // clear ANDES
+    hybrid_state_v1 got = ref;
+    got.andes.uitb = 0xbad;
+    got.andes.pmaaddr[0] = 0xbad;
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // msp_bound/msp_base are engine-reset-divergent (QEMU resets them to
+     // all-ones; a vsim core without HW stack protection drains 0), so verify
+     // EXCLUDES them like mip -- still transported, just not strict-compared.
+    const hybrid_state_v1 ref = andes_state();
+    hybrid_state_v1 got = ref;
+    got.andes.msp_bound = 0;
+    got.andes.msp_base = 0;
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // milmb/mdlmb are the ILM/DLM base+enable CSRs. The base address matches
+     // across engines, but the enable bit (bit 0) is HW-presence state: the vsim
+     // RTL has the local memories and drains EN=1, while QEMU models none and
+     // drains EN=0. Same engine-reset-divergent class as msp_bound/msp_base, so
+     // verify EXCLUDES them -- still transported, just not strict-compared.
+    const hybrid_state_v1 ref = andes_state();
+    hybrid_state_v1 got = ref;
+    got.andes.milmb ^= 0x1;  // flip the enable bit only (base unchanged)
+    got.andes.mdlmb ^= 0x1;
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // mnvec is engine-reset-divergent: QEMU resets it to 0, the Andes RTL
+     // resets it to the hart's reset vector (0x100000 seen live on both
+     // engines) and WARL-holds it, so a trap-free ROI always false-diffs.
+     // Same class as msp_bound/msp_base: transported, not strict-compared.
+    hybrid_state_v1 ref = andes_state();
+    ref.flags |= HYBRID_FLAG_M_EXT;
+    hybrid_state_v1 got = ref;
+    got.m_ext.mnvec = 0x100000;                  // engine reset vector
+    CHECK(diff_states(ref, got).ok());
+    got.m_ext.mcountinhibit = 0xbad;             // control: M-ext IS compared
+    CHECK(!diff_states(ref, got).ok());
+  }
+
+  {  // S4 Andes caps: a CSR the engine does NOT implement is verify-excluded.
+     // vsim never restored/re-read it (the resume/drain gate), so comparing it
+     // against the QEMU oracle would false-diff. The supported list comes from
+     // the engine's --describe andes_csrs.
+    const hybrid_state_v1 ref = andes_state();
+    hybrid_state_v1 got = ref;
+    got.andes.ucode ^= 0xDEAD;
+
+    // engine lacks ucode -> excluded, no diff.
+    CHECK(diff_states(ref, got, {"uitb", "milmb"}).ok());
+    // engine has ucode -> the diff IS reported.
+    const DiffReport in = diff_states(ref, got, {"ucode"});
+    CHECK(in.diffs.size() == 1);
+    CHECK(in.diffs[0].field == "ucode");
+    // empty list (default) = compare all (state-diff subcommand / no engine info).
+    CHECK(diff_states(ref, got).diffs.size() == 1);
+  }
+
+  {  // S4 Andes caps: the PMA block is excluded via the "pmaaddr"/"pmacfg" markers
+    const hybrid_state_v1 ref = andes_state();
+    hybrid_state_v1 got = ref;
+    got.andes.pmaaddr[0] ^= 0xDEAD;
+    CHECK(diff_states(ref, got, {"uitb"}).ok());  // no PMA support -> excluded
+    const DiffReport in = diff_states(ref, got, {"pmaaddr"});
+    CHECK(in.diffs.size() == 1);
+    CHECK(in.diffs[0].field == "pmaaddr0");
+  }
+
+  {  // v5: L2C Control Register RW bits round-trip (restored at resume, re-read
+     // at drain), so the hybrid policy compares them and names a mismatch.
+    hybrid_state_v1 ref{};
+    ref.magic = HYBRID_STATE_MAGIC;
+    ref.version = HYBRID_STATE_VERSION;
+    ref.flags = HYBRID_FLAG_RV64 | HYBRID_FLAG_L2C;
+    ref.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0x1;  // CEN=1
+    hybrid_state_v1 got = ref;
+    got.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0x0;  // transport lost the enable
+    const DiffReport r = diff_states(ref, got);
+    CHECK(!r.ok());
+    CHECK(r.diffs.size() == 1);
+    CHECK(r.diffs[0].field == "l2c.ctrl");
+  }
+
+  {  // v5: INITSTATUS[14] is read-only status (a real L2C model drives it, the
+     // QEMU RAM stub does not), so it is masked out of the compare.
+    hybrid_state_v1 ref{};
+    ref.magic = HYBRID_STATE_MAGIC;
+    ref.version = HYBRID_STATE_VERSION;
+    ref.flags = HYBRID_FLAG_RV64 | HYBRID_FLAG_L2C;
+    ref.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0x1;                    // INITSTATUS=0
+    hybrid_state_v1 got = ref;
+    got.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0x1 | (1ull << 14);     // INITSTATUS=1
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // v5: with HYBRID_FLAG_L2C clear the L2C block is not live, so it is skipped.
+    hybrid_state_v1 ref{};
+    ref.magic = HYBRID_STATE_MAGIC;
+    ref.version = HYBRID_STATE_VERSION;
+    ref.flags = HYBRID_FLAG_RV64;  // no L2C flag
+    ref.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0x1;
+    hybrid_state_v1 got = ref;
+    got.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8] = 0xdead;  // differs, but not compared
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // v6: with a write-mask, an UNWRITTEN CSR is not compared -- its drained
+     // value is only QEMU's reset default, which may legitimately differ from
+     // the engine's RTL reset value the re-read returns.
+    hybrid_state_v1 ref = andes_state();
+    ref.flags |= HYBRID_FLAG_WRITE_MASK;   // mask live, all bits clear
+    hybrid_state_v1 got = ref;
+    got.andes.mxstatus ^= 0xF00;           // reset-divergent, but unwritten
+    got.andes.pmaaddr[2] ^= 0xF00;
+    CHECK(diff_states(ref, got).ok());
+  }
+
+  {  // v6: a WRITTEN CSR still compares -- and a mismatch is named.
+    hybrid_state_v1 ref = andes_state();
+    ref.flags |= HYBRID_FLAG_WRITE_MASK;
+    hybrid_wm_set(ref.write_mask,
+                  HYBRID_WM_ANDES_BASE + HYBRID_WM_ANDES_IDX_mxstatus);
+    hybrid_state_v1 got = ref;
+    got.andes.mxstatus ^= 0xF00;
+    const DiffReport r = diff_states(ref, got);
+    CHECK(!r.ok());
+    CHECK(r.diffs.size() == 1);
+    CHECK(r.diffs[0].field == "mxstatus");
+  }
+
+  {  // v6: flag clear (pre-v6 file) -> compare everything, the old behavior.
+    hybrid_state_v1 ref = andes_state();   // no WRITE_MASK flag
+    hybrid_state_v1 got = ref;
+    got.andes.mxstatus ^= 0xF00;
+    const DiffReport r = diff_states(ref, got);
+    CHECK(!r.ok());
+    CHECK(r.diffs.size() == 1);
+    CHECK(r.diffs[0].field == "mxstatus");
+  }
+
+  {  // v6: unsupported_written_csrs -- "program wrote it, engine lacks it" is a
+     // REAL incompatibility signal (on that engine's silicon the csrw would
+     // trap at boot), so the driver surfaces it instead of silently skipping.
+    hybrid_state_v1 st{};
+    st.magic = HYBRID_STATE_MAGIC;
+    st.version = HYBRID_STATE_VERSION;
+    st.flags = HYBRID_FLAG_RV64 | HYBRID_FLAG_ANDES | HYBRID_FLAG_WRITE_MASK;
+    hybrid_wm_set(st.write_mask,
+                  HYBRID_WM_ANDES_BASE + HYBRID_WM_ANDES_IDX_mxstatus);
+    hybrid_wm_set(st.write_mask,
+                  HYBRID_WM_ANDES_BASE + HYBRID_WM_ANDES_IDX_msp_bound);
+    hybrid_wm_set(st.write_mask, HYBRID_WM_PMAADDR_BASE + 3);
+    // ucode's bit is ALWAYS set by the plugin (transport-always DSP flag), so
+    // it must never be reported even when the engine lacks it.
+    hybrid_wm_set(st.write_mask,
+                  HYBRID_WM_ANDES_BASE + HYBRID_WM_ANDES_IDX_ucode);
+
+    // Engine implements mxstatus but not msp_bound / the PMA block / ucode.
+    const std::vector<std::string> caps = {"mxstatus", "jvt"};
+    const auto bad = unsupported_written_csrs(st, caps);
+    CHECK(bad.size() == 2);
+    bool has_msp = false, has_pma = false;
+    for (const auto& n : bad) {
+      if (n == "msp_bound") has_msp = true;
+      if (n == "pmaaddr") has_pma = true;
+    }
+    CHECK(has_msp);
+    CHECK(has_pma);
+
+    // Written + supported -> clean.
+    const std::vector<std::string> full = {"mxstatus", "msp_bound", "pmaaddr"};
+    CHECK(unsupported_written_csrs(st, full).empty());
+
+    // No engine info (empty list) -> no report (nothing to check against).
+    CHECK(unsupported_written_csrs(st, {}).empty());
+
+    // Pre-v6 file (flag clear) -> no report (no write information).
+    hybrid_state_v1 old = st;
+    old.flags &= ~HYBRID_FLAG_WRITE_MASK;
+    CHECK(unsupported_written_csrs(old, caps).empty());
+
+    // The driver-facing warning: one stderr-ready line naming the engine and
+    // the offending CSRs, or empty when there is nothing to report.
+    const std::string w = unsupported_written_warning(st, caps, "vsim:eng");
+    CHECK(w.find("warning") != std::string::npos);
+    CHECK(w.find("vsim:eng") != std::string::npos);
+    CHECK(w.find("msp_bound") != std::string::npos);
+    CHECK(w.find("pmaaddr") != std::string::npos);
+    CHECK(w.find("mxstatus") == std::string::npos);  // supported: not listed
+    CHECK(!w.empty() && w.back() == '\n');
+    CHECK(unsupported_written_warning(st, full, "vsim:eng").empty());
+    CHECK(unsupported_written_warning(old, caps, "vsim:eng").empty());
+  }
+
   return andes_sim::test::report("state_diff");
 }
