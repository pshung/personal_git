  29 class ResumeDriver
  30 {
  31 public:
  32   // data0/data1/abstractcs/command are defined once in abstract_command.hpp;
  33   // dmcontrol/dmstatus are run-control registers specific to this driver.

  35   static constexpr uint8_t kData1      = kDmiData1;
  36   static constexpr uint8_t kDmcontrol  = 0x10;
  37   static constexpr uint8_t kDmstatus   = 0x11;
  38   static constexpr uint8_t kAbstractcs = kDmiAbstractcs;
  39   static constexpr uint8_t kCommand    = kDmiCommand;

  41   static constexpr uint16_t kCsrDpc      = 0x7B1;
  42   static constexpr uint16_t kCsrDcsr     = 0x7B0;
  43   static constexpr uint16_t kCsrPmpcfg0  = 0x3A0;
  44   static constexpr uint16_t kCsrPmpaddr0 = 0x3B0;
  45   static constexpr uint16_t kCsrFcsr     = 0x003;
  46   static constexpr uint16_t kCsrMstatus  = 0x300;
  47   static constexpr uint16_t kCsrMisa     = 0x301;
  48   static constexpr uint16_t kCsrMedeleg  = 0x302;
  49   static constexpr uint16_t kCsrMideleg  = 0x303;
  50   static constexpr uint16_t kCsrMie      = 0x304;
  51   static constexpr uint16_t kCsrMtvec    = 0x305;
  52   static constexpr uint16_t kCsrMscratch = 0x340;
  53   static constexpr uint16_t kCsrMepc     = 0x341;

  59   static constexpr uint16_t kCsrMinstret = 0xB02;
  60   // Andes machine setup CSRs crt0 writes at boot (see hybrid_state_v1):
  61   // mcache_ctl enables the L1 I/D caches, mmisc_ctl holds misc machine config.
  62   static constexpr uint16_t kCsrMcacheCtl = 0x7CA;
  63   static constexpr uint16_t kCsrMmiscCtl  = 0x7D0;
  64   static constexpr uint16_t kCsrVlenb    = 0xC22;
  65   static constexpr uint16_t kCsrVstart   = 0x008;
  66   static constexpr uint16_t kCsrVxsat    = 0x009;
  67   static constexpr uint16_t kCsrVxrm     = 0x00A;
  68   static constexpr uint16_t kCsrVcsr     = 0x00F;
  69   static constexpr uint16_t kCsrVtype    = 0xC21;
  70   static constexpr uint16_t kCsrVl       = 0xC20;

  72   static constexpr uint32_t kDmcontrolDmactive  = 1u << 0;
  73   static constexpr uint32_t kDmcontrolResumereq = 1u << 30;
  74   static constexpr uint32_t kDmcontrolHaltreq   = 1u << 31;
  75 
  76   static constexpr uint32_t kDmstatusAllhalted    = 1u << 9;
  77   static constexpr uint32_t kDmstatusAllresumeack = 1u << 17;
  78 
  79   static constexpr uint32_t kAbstractcsBusy        = hybrid::kAbstractcsBusy;
  80   static constexpr uint32_t kAbstractcsCmderrMask  = hybrid::kAbstractcsCmderrMask;
  81 
  82   // mstatus extension-state fields: FS (bits 14:13), VS (bits 10:9).
  83   // "Initial" (01) is the minimal enable the abstract FPR / V writes require.
  84   static constexpr uint64_t kMstatusFsMask    = 0x6000ULL;
  85   static constexpr uint64_t kMstatusFsInitial = 0x2000ULL;
  86   static constexpr uint64_t kMstatusVsMask    = 0x600ULL;
  87   static constexpr uint64_t kMstatusVsInitial = 0x200ULL;
  88 
  89   static constexpr unsigned kPollIterations = kDmiPollIterations;

  91   // scratch_gpa: the GUEST address of the V-register transport page (L26:
  92   // v_scratch_gpa(ram_base, ram_size) = the mmap's top page). Defaults to the
  93   // "unset" sentinel; a non-V resume never reads it, but a V resume with the
  94   // GPA still unset throws in the sweep (no silent transport via a bogus
  95   // address). The simulator passes the real value from platform.main_memory.
  96   // caps: which Andes custom CSRs the TARGET core implements (derived from its
  97   // config.inc, supplied by the sim binary). Defaults to all() -- transport
  98   // every CSR, the prior behavior -- so existing call sites are unchanged.
  99   explicit ResumeDriver(DmiBus& bus, uint64_t scratch_gpa = kVScratchGpaUnset,
 100                         AndesCaps caps = AndesCaps::all())
 101       : bus_{bus}, scratch_gpa_{scratch_gpa}, caps_{caps}
 102   {}
 103 
 104   // LG1/LG2 (liveness gate): supply the engine's own reset-state drain.
 105   // Restores whose drained value equals this baseline are skipped -- the
 106   // write would be a no-op, and every DM operation costs hundreds of
 107   // simulated cycles. LG1 gates each tolerant CSR write; LG2 elides the FPR
 108   // and V data blocks as whole blocks when byte-equal. Unset (default) keeps
 109   // transport-everything. Always-live state (mstatus, GPRs, dpc, dcsr,
 110   // mcycle, minstret, the S/U PMP mirror) is never gated.
 111   //
 112   // v_cfg_at_reset: pass false when the baseline was CAPTURED this run --
 113   // the capture drain's V dump sweep already clobbered the hart's
 114   // vtype/vl/vstart, so their baseline equality no longer proves the
 115   // architectural vsetvl redundant. Cache-hit runs (pristine hart) pass true.
 116   void set_reset_baseline(const hybrid_state_v1& b, bool v_cfg_at_reset = true)
 117   {
 118     baseline_ = b;
 119     has_baseline_ = true;
 120     v_cfg_at_reset_ = v_cfg_at_reset;
 121   }
 122 
 123   // Number of register-restore operations elided by the reset baseline this
 124   // resume: one per gated CSR write, 32 per elided FPR/V data block, one per
 125   // elided vsetvl.
 126   auto baseline_skip_count() const -> uint64_t { return baseline_skip_count_; }
 127 
 128   // v5: declare whether the target engine has an L2 cache controller, and its
 129   // MMIO register base. Off by default so an engine without an L2C (every
 130   // current build: NDS_L2C_CACHE_SIZE_KB=0) issues NO L2C store even if the
 131   // handoff carries the block -- a store to an absent region would fault the
 132   // Program Buffer, unlike a tolerant CSR write. The simulator calls this from
 133   // its engine config (has_shared_cache + NDS_L2C_REG_BASE).
 134   void set_l2c(bool present, uint64_t base)
 135   {
 136     l2c_present_ = present;
 137     l2c_base_    = base;
 138   }

 140   void resume(const hybrid_state_v1& state)
 141   {
 142     if (state.magic != HYBRID_STATE_MAGIC)
 143       throw std::runtime_error("ResumeDriver::resume: bad magic");
 144     if (state.version != HYBRID_STATE_VERSION)
 145       throw std::runtime_error("ResumeDriver::resume: unsupported version");
 146     // L6: only U(0)/S(1)/M(3) fit dcsr.prv; 2 is the reserved encoding.
 147     if (state.priv == 2 || state.priv > 3)
 148       throw std::runtime_error("ResumeDriver::resume: invalid privilege");
 149 
 150     // dmactive must transition 0->1 in its own write before any other
 151     // DMCONTROL bit is honored (RISC-V Debug Spec).
 152     bus_.write(kDmcontrol, kDmcontrolDmactive);
 153     request_halt();
 154     poll_until_set(bus_, kDmstatus, kDmstatusAllhalted, "ResumeDriver halt");
 155 
 156     // mstatus: enable FS / VS so the following FPR / V abstract writes are
 157     // accepted (FS or VS == Off disables the extension and the hart NACKs
 158     // the writes). The drained state.mstatus may legitimately carry FS or
 159     // VS at Off (e.g. an OS context-switched the regs away before the
 160     // trigger fired) -- gating the write on the drained field skips the
 161     // enable and silently NACKs every subsequent FPR / V-CSR write. Always
 162     // write mstatus when either extension is in scope; force-OR Initial
 163     // when the drained field is Off. This enable is local to the vsim
 164     // phase; the state.mstatus written to the handoff file is left
 165     // unmodified.
 166     if (state.flags & (HYBRID_FLAG_FP | HYBRID_FLAG_V)) {
 167       uint64_t enable_mstatus = state.mstatus;
 168       if ((state.flags & HYBRID_FLAG_FP) && (enable_mstatus & kMstatusFsMask) == 0)
 169         enable_mstatus |= kMstatusFsInitial;  // FS = Initial
 170       if ((state.flags & HYBRID_FLAG_V)  && (enable_mstatus & kMstatusVsMask) == 0)
 171         enable_mstatus |= kMstatusVsInitial;  // VS = Initial
 172       write_csr_abstract_or_pb(kCsrMstatus, enable_mstatus);
 173     }
 174 
 175     // FPR restore (requires mstatus.FS enabled above): one abstract register
 176     // write per f-reg. Abstract-only by design, like every non-V register
 177     // block -- the Program Buffer is reserved for state the abstract command
 178     // space cannot reach (the V regfile, vtype/vl, L2C MMIO).
 179     if (state.flags & HYBRID_FLAG_FP) {
 180       // LG2: elide the whole FPR block when every f-reg already holds the
 181       // engine's reset value -- the reload would rewrite what is there.
 182       const bool fpr_equal = has_baseline_
 183         && std::memcmp(state.fpr, baseline_.fpr, sizeof state.fpr) == 0;
 184       if (fpr_equal) {
 185         baseline_skip_count_ += 32;
 186       } else {
 187         for (uint16_t i = 0; i < 32; ++i)
 188           write_64bit_register(regno_fpr(i), state.fpr[i]);
 189       }
 190       // L20: fcsr (frm + sticky fflags). Same FS gate as the FPRs; PB
 191       // fallback scratch x5 is overwritten by the GPR loop below.
 192       if (!baseline_skips(state.fcsr, baseline_.fcsr))
 193         write_csr_abstract_or_pb(kCsrFcsr, state.fcsr);
 194     }
 195 
 196     // Tier-A V CSR restore (before GPRs: PB fallback uses x5 as scratch;
 197     // the GPR loop overwrites x5 afterward so state is preserved).
 198     if (state.flags & HYBRID_FLAG_V) {
 199       // V data regfile: reload v0..v31 from the shared scratch region via the
 200       // Program Buffer. Runs BEFORE the architectural V-CSR restore below
 201       // because the sweep's own vsetvl (e8/m8) clobbers vtype/vl/vstart, which
 202       // the restore then overwrites with the architectural values. The sweep
 203       // uses x5/x6/x7 as scratch (x7 is the group base pointer); the GPR loop
 204       // below restores all three. The scratch is vlenb-packed
 205       // (stage_v_to_scratch), so the sweep takes only the runtime vlenb.
 206       if (scratch_gpa_ == kVScratchGpaUnset)
 207         throw std::runtime_error(
 208           "ResumeDriver::resume: V state present but scratch GPA unset");
 209       // LG2: elide the reload sweep when every v-reg already holds the
 210       // engine's reset bytes.
 211       const bool v_data_equal = has_baseline_
 212         && std::memcmp(state.vreg, baseline_.vreg, sizeof state.vreg) == 0;
 213       if (v_data_equal)
 214         baseline_skip_count_ += 32;
 215       else
 216         pb_v_data_sweep(bus_, scratch_gpa_, state.vlenb, /*load=*/true);
 217 
 218       // Tier-B: vtype+vl via vsetvl x0,x5,x6 in PB.
 219       // Direct csrrw to vtype sets vill; vl is read-only. Stage operands in
 220       // scratch GPRs x5(vl) and x6(vtype) then let vsetvl set both atomically.
 221       // x5/x6 are overwritten by the GPR loop below, so no leakage.
 222       // LG2 clobber rule: the data sweep's own vsetvl overwrote vtype/vl/
 223       // vstart, so baseline equality only proves this vsetvl redundant when
 224       // the sweep did NOT run.
 225       const bool vcfg_equal = v_data_equal && v_cfg_at_reset_
 226         && state.vl == baseline_.vl && state.vtype == baseline_.vtype;
 227       if (vcfg_equal) {
 228         ++baseline_skip_count_;  // the elided vsetvl exec
 229       } else {
 230         write_64bit_register(regno_gpr(5), state.vl);
 231         write_64bit_register(regno_gpr(6), state.vtype);
 232         const std::array<uint32_t, 1> pb_vsetvl = { encode_vsetvl(0, 5, 6) };
 233         exec_progbuf(bus_, std::span<const uint32_t>{pb_vsetvl});
 234       }
 235       // vstart must come AFTER vsetvl: vsetvl resets vstart to 0 -- so its
 236       // baseline skip is likewise only valid when no vsetvl executed.
 237       if (!(vcfg_equal && baseline_skips(state.vstart, baseline_.vstart)))
 238         write_csr_abstract_or_pb(kCsrVstart, state.vstart);
 239       if (!baseline_skips(state.vxsat, baseline_.vxsat))
 240         write_csr_abstract_or_pb(kCsrVxsat,  state.vxsat);
 241       if (!baseline_skips(state.vxrm, baseline_.vxrm))
 242         write_csr_abstract_or_pb(kCsrVxrm,   state.vxrm);
 243       if (!baseline_skips(state.vcsr, baseline_.vcsr))
 244         write_csr_abstract_or_pb(kCsrVcsr,   state.vcsr);
 245     }
 246 
 247     // L6: dret resumes into dcsr.prv, which is M after the boot halt. For an
 248     // S/U-mode handoff, read-modify-write the prv field so the ROI executes
 249     // in the drained privilege; M-mode states skip the write (prv already 3).
 250     // Before the GPR loop: a PB fallback would clobber scratch x5.
 251     if (state.priv != 3) {
 252       // L2 slice: with PMP implemented and no matching entry, every S/U
 253       // access faults (priv spec v1.10) -- mirror the producer's PMP block
 254       // so the ROI can fetch at all. All pmpaddr before any pmpcfg (a locked
 255       // cfg discards later addr writes); odd pmpcfg CSRs do not exist on
 256       // RV64. Tolerant writes: entries beyond what the hart implements
 257       // mirror as no-ops. M-mode handoffs keep PMP pass-through-only (L2).
 258       for (uint16_t i = 0; i < HYBRID_PMP_ADDR_NUM; ++i)
 259         write_csr_tolerant(kCsrPmpaddr0 + i, state.pmpaddr[i]);
 260       for (uint16_t i = 0; i < HYBRID_PMP_CFG_NUM; i += 2)
 261         write_csr_tolerant(kCsrPmpcfg0 + i, state.pmpcfg[i]);
 262 
 263       const uint64_t dcsr =
 264         read_register_64(bus_, regno_csr(kCsrDcsr), "ResumeDriver dcsr");
 265       write_csr_abstract_or_pb(kCsrDcsr,
 266                                (dcsr & ~uint64_t{0x3}) | state.priv);
 267       // An M-only core WARL-pins dcsr.prv to 3; resuming anyway would
 268       // silently measure the ROI in the wrong privilege. Verify and abort.
 269       const uint64_t got =
 270         read_register_64(bus_, regno_csr(kCsrDcsr), "ResumeDriver dcsr verify")
 271         & 0x3u;
 272       if (got != state.priv)
 273         throw std::runtime_error(
 274           "ResumeDriver::resume: engine does not implement privilege level "
 275           + std::to_string(state.priv)
 276           + " (dcsr.prv pinned to " + std::to_string(got)
 277           + "); pick an S/U-capable engine");
 278     }
 279 
 280     // Andes V5 + M-ext CSR restore (abstract-only, LG1-gated). See
 281     // restore_andes_m_ext.
 282     restore_andes_m_ext(state);
 283 
 284     // v5: L2C MMIO config restore. Runs BEFORE the GPR loop (like the V sweep):
 285     // it stages an address + value into scratch GPRs x5/x6 and the GPR loop
 286     // below overwrites them with their architectural values, so the clobber is
 287     // safe. It CANNOT run after the loop, where every GPR already holds its
 288     // final value. See restore_l2c.
 289     restore_l2c(state);
 290 
 291     // GPRs + DPC (after V CSRs so PB scratch x5 is safe to overwrite): one
 292     // abstract register write per GPR.
 293     for (uint16_t i = 1; i < 32; ++i)
 294       write_64bit_register(regno_gpr(i), state.gpr[i]);
 295     write_64bit_register(regno_csr(kCsrDpc), state.pc);
 296 
 297     write_csr_tolerant_gated(kCsrMtvec, state.mtvec, baseline_.mtvec);
 298 
 299     // L23: mscratch is the M-mode software scratch CSR -- a pure read/write
 300     // register with no hardware side effects, legal to use inside a trap-free
 301     // ROI. Restore it so a value stashed before the handoff survives into the
 302     // vsim phase.
 303     write_csr_tolerant_gated(kCsrMscratch, state.mscratch, baseline_.mscratch);
 304 
 305     // L24: the writable M-mode trap/interrupt CSRs. In the trap-free,
 306     // interrupt-free ROI envelope these are ordinary read/write registers, so a
 307     // value stashed before the handoff must survive into the vsim phase (same
 308     // round-trip as mscratch). Tolerant writes: a WARL / read-only field (mip's
 309     // hardware-driven pending bits, or mtval on a core that hardwires it)
 310     // mirrors as a no-op. medeleg/mideleg/satp stay unrestored -- they are
 311     // S-mode CSRs the M-only default engine does not implement (L3/L4).
 312     write_csr_tolerant_gated(kCsrMepc,   state.mepc,   baseline_.mepc);
 313     write_csr_tolerant_gated(kCsrMcause, state.mcause, baseline_.mcause);
 314     write_csr_tolerant_gated(kCsrMtval,  state.mtval,  baseline_.mtval);
 315     write_csr_tolerant_gated(kCsrMie,    state.mie,    baseline_.mie);
 316     write_csr_tolerant_gated(kCsrMip,    state.mip,    baseline_.mip);
 317 
 318     // Andes cache fix: restore the machine setup CSRs crt0 wrote at boot. The
 319     // functional engine ran crt0 (where `csrw mcache_ctl` enables the L1 I/D
 320     // caches) but has no cache model, so without this the cycle engine resumes
 321     // at the ROI with caches at their reset default -- DISABLED -- and the
 322     // whole measured window runs uncached, inflating it severalfold. Tolerant
 323     // writes: an engine without these custom CSRs (e.g. non-Andes) mirrors the
 324     // write as a no-op, and a drained 0 (caches were off) restores the reset
 325     // state, so this is always safe to issue.
 326     if (wm_live(state, HYBRID_WM_MCACHE_CTL))
 327       write_csr_tolerant_gated(kCsrMcacheCtl, state.mcache_ctl,
 328                                baseline_.mcache_ctl);
 329     if (wm_live(state, HYBRID_WM_MMISC_CTL))
 330       write_csr_tolerant_gated(kCsrMmiscCtl,  state.mmisc_ctl,
 331                                baseline_.mmisc_ctl);
 332 
 333     // F-T2.4: restore mcycle/minstret so the vsim phase continues counting
 334     // from QEMU1's drain value. Without this, vsim's local counter starts
 335     // at zero, so the drained mcycle delta (vsim minus QEMU 1) would no
 336     // longer measure the ROI. PB fallback uses x5 as scratch; safe to
 337     // overwrite here as the GPR loop already ran.
 338     write_csr_abstract_or_pb(kCsrMcycle,   state.mcycle);
 339     write_csr_abstract_or_pb(kCsrMinstret, state.minstret);
 340 
 341     clear_cmderr_if_set();
 342     request_resume();
 343     poll_until_set(bus_, kDmstatus, kDmstatusAllresumeack,
 344                    "ResumeDriver resume");
 345   }

 348   void request_halt()
 349   {
 350     bus_.write(kDmcontrol, kDmcontrolDmactive | kDmcontrolHaltreq);

 353   void request_resume()

 355     bus_.write(kDmcontrol, kDmcontrolDmactive | kDmcontrolResumereq);

 358   void write_64bit_register(uint16_t regno, uint64_t value)
 359   {
 360     write_register_abstract(bus_, regno, value);
 361   }

 363   // v5: write back the WRITABLE boot-static L2C config registers via
 364   // Program-Buffer word stores. The L2C block is MMIO, not a CSR, so each write
 365   // is `sw x5, 0(x6)` with the register address staged in x6 and the value (low
 366   // word) in x5. Restores ONLY the config registers the ROI can hold as plain
 367   // state: Control (CEN/prefetch/ECC/RAM timing), HPM Control, Way Allocation
 368   // Masks. The read-only Configuration Register, the side-effecting CCTL command
 369   // registers (writing one re-issues an L2 cache op), and the write-1-clear
 370   // error registers are captured in the record but NEVER written back.
 371   //
 372   // Hard-gated on the engine having an L2C (l2c_present_): a store to an absent
 373   // L2C region faults the Program Buffer, so unlike a tolerant CSR write this
 374   // must issue nothing on a core without an L2C. x5/x6 are safe scratch here --
 375   // the caller runs this before the GPR loop, which overwrites both.
 376   void restore_l2c(const hybrid_state_v1& state)
 377   {
 378     if (!(state.flags & HYBRID_FLAG_L2C)) return;
 379     if (!l2c_present_) return;
 380 
 381     pb_store_l2c_reg(HYBRID_L2C_OFF_CONTROL,
 382                      state.l2c.reg[HYBRID_L2C_OFF_CONTROL / 8]);
 383     pb_store_l2c_reg(HYBRID_L2C_OFF_HPM_CTL0,
 384                      state.l2c.reg[HYBRID_L2C_OFF_HPM_CTL0 / 8]);
 385     for (uint32_t i = 0; i < HYBRID_L2C_WAYMASK_NUM; ++i) {
 386       const uint32_t off = HYBRID_L2C_OFF_WAYMASK0 + i * 8u;
 387       pb_store_l2c_reg(off, state.l2c.reg[off / 8]);
 388     }
 389   }
 390 
 391   // One L2C register store: stage the MMIO address in x6 and the value's low
 392   // word in x5, then execute `sw x5, 0(x6)` in the Program Buffer. Word store
 393   // per DS220 11.8.3 (single transfer, access size < 8 bytes); all documented
 394   // L2C control bits live in the low 32.
 395   void pb_store_l2c_reg(uint32_t offset, uint64_t value)
 396   {
 397     write_register_abstract(bus_, regno_gpr(6), l2c_base_ + offset);
 398     write_register_abstract(bus_, regno_gpr(5), value & 0xffffffffu);
 399     const std::array<uint32_t, 1> pb = { encode_sw(5, 6, 0) };
 400     exec_progbuf(bus_, std::span<const uint32_t>{pb});
 401   }
 402 
 403   // Andes V5 + M-ext CSR restore: one caps-gated abstract write per CSR
 404   // (pmaaddr before pmacfg, mirroring the PMP restore ordering), each gated by
 405   // the LG1 reset baseline. Abstract-only by design -- a batched Program-Buffer
 406   // sweep can't tolerate a cmderr on one CSR without corrupting the rest of the
 407   // batch (a WARL/read-only CSR faults the whole group mid-sweep), whereas the
 408   // per-CSR abstract path tolerates a single failure and moves on.
 409   void restore_andes_m_ext(const hybrid_state_v1& state)
 410   {
 411     const bool andes = state.flags & HYBRID_FLAG_ANDES;
 412     const bool mext  = state.flags & HYBRID_FLAG_M_EXT;
 413     if (!andes && !mext) return;
 414 
 415     // Each write is double-gated: caps (the ENGINE implements the CSR) and the
 416     // v6 write-mask (the PRODUCER's guest actually wrote it -- wm_live).
 417     if (andes) {
 418 #define X(field, addr, name)                                                  \
 419   if (caps_.supports_csr(addr)                                                \
 420       && wm_live(state, HYBRID_WM_ANDES_BASE + HYBRID_WM_ANDES_IDX_##field))  \
 421     write_csr_tolerant_gated(addr, state.andes.field, baseline_.andes.field);
 422       HYBRID_ANDES_CSR_LIST(X)
 423 #undef X
 424       for (uint16_t i = 0; i < HYBRID_PMA_ADDR_NUM; ++i)
 425         if (caps_.supports_csr(HYBRID_PMA_ADDR0_CSR + i)
 426             && wm_live(state, HYBRID_WM_PMAADDR_BASE + i))
 427           write_csr_tolerant_gated(HYBRID_PMA_ADDR0_CSR + i,
 428                                    state.andes.pmaaddr[i],
 429                                    baseline_.andes.pmaaddr[i]);
 430       for (uint16_t i = 0; i < HYBRID_PMA_CFG_NUM; ++i)
 431         if (caps_.supports_csr(HYBRID_PMA_CFG0_CSR + 2 * i)
 432             && wm_live(state, HYBRID_WM_PMACFG_BASE + i))
 433           write_csr_tolerant_gated(HYBRID_PMA_CFG0_CSR + 2 * i,
 434                                    state.andes.pmacfg[i],
 435                                    baseline_.andes.pmacfg[i]);
 436     }
 437     if (mext) {
 438 #define X(field, addr, name)                                                  \
 439   if (caps_.supports_csr(addr)                                                \
 440       && wm_live(state, HYBRID_WM_M_EXT_BASE + HYBRID_WM_M_EXT_IDX_##field))  \
 441     write_csr_tolerant_gated(addr, state.m_ext.field, baseline_.m_ext.field);
 442       HYBRID_M_EXT_CSR_LIST(X)
 443 #undef X
 444     }
 445   }
 446 
 447   // Abstract-first CSR write with Program Buffer fallback.
 448   // Tries an abstract register write; if the DM signals cmderr, clears it,
 449   // stages the value in scratch_gpr via abstract GPR write, then executes a
 450   // one-instruction PB (`csrrw x0, csr, scratch_gpr`).  scratch_gpr defaults
 451   // to x5 (t0) which the GPR restore loop overwrites later.
 452   void write_csr_abstract_or_pb(uint16_t csr, uint64_t value,
 453                                  uint8_t scratch_gpr = 5)
 454   {
 455     write_64bit_register(regno_csr(csr), value);
 456     if ((bus_.read(kAbstractcs) & kAbstractcsCmderrMask) == 0u) return;
 457     bus_.write(kAbstractcs, kAbstractcsCmderrMask);  // W1C: clear sticky error
 458     write_64bit_register(regno_gpr(scratch_gpr), value);
 459     const std::array<uint32_t, 1> pb = { encode_csrrw(0, csr, scratch_gpr) };
 460     exec_progbuf(bus_, std::span<const uint32_t>{pb});
 461   }
 462 
 463   // Write a CSR-addressable register and clear any cmderr the DM raised.
 464   // Used for CSRs whose abstract-access support may be implementation
 465   // dependent; a missing CSR mirrors as a no-op rather than blocking the
 466   // remaining writes.
 467   void write_csr_tolerant(uint16_t csr, uint64_t value)
 468   {
 469     write_64bit_register(regno_csr(csr), value);
 470     clear_cmderr_if_set();
 471   }
 472 
 473   // v6: true when the handoff says this mask-governed CSR is live -- the
 474   // producer's guest actually wrote it, or the producer carried no mask at all
 475   // (pre-v6 file: restore-everything fallback). An untouched CSR is skipped so
 476   // the engine keeps its OWN reset value: the drained value is only the
 477   // producer's reset default, which may legitimately differ from this RTL's
 478   // (msp_bound all-ones vs 0, milmb.IEN 0 vs 1; see hybrid/write_mask.h).
 479   static auto wm_live(const hybrid_state_v1& state, int slot) -> bool
 480   {
 481     if (!(state.flags & HYBRID_FLAG_WRITE_MASK)) return true;
 482     return hybrid_wm_test(state.write_mask, static_cast<unsigned>(slot)) != 0;
 483   }
 484 
 485   // LG1: true (and counted) when the reset baseline marks this write a no-op.
 486   auto baseline_skips(uint64_t value, uint64_t reset_value) -> bool
 487   {
 488     if (!has_baseline_ || value != reset_value) return false;
 489     ++baseline_skip_count_;
 490     return true;
 491   }
 492 
 493   // Tolerant CSR write elided when the value already equals the engine's
 494   // reset value (reset_value is only meaningful when the baseline is set;
 495   // baseline_ is zero-initialized otherwise and baseline_skips ignores it).
 496   void write_csr_tolerant_gated(uint16_t csr, uint64_t value,
 497                                 uint64_t reset_value)
 498   {
 499     if (baseline_skips(value, reset_value)) return;
 500     write_csr_tolerant(csr, value);
 501   }
 502 
 503   // abstractcs.cmderr is sticky per the Debug Spec - a subsequent command
 504   // is rejected until the bits are W1C-cleared. We tolerate per-register
 505   // failures (e.g. unimplemented FPR/CSR Abstract access) by clearing the
 506   // error before resuming so the hart isn't left in a stuck state.
 507   // B17: record the last observed non-zero cmderr and a clear count so a
 508   // partially-broken restore is observable via the public accessors
 509   // instead of silently swallowed.
 510   void clear_cmderr_if_set()
 511   {
 512     const auto status = bus_.read(kAbstractcs);
 513     const auto code = (status & kAbstractcsCmderrMask) >> 8;
 514     if (code != 0u) {
 515       last_observed_cmderr_ = code;
 516       ++cmderr_clear_count_;
 517       bus_.write(kAbstractcs, kAbstractcsCmderrMask);
 518     }
 519   }
 520 
 521   DmiBus& bus_;
 522   uint64_t scratch_gpa_ = kVScratchGpaUnset;  // L26: V transport page GPA
 523   AndesCaps caps_ = AndesCaps::all();         // target core's Andes-CSR support
 524   bool l2c_present_ = false;                  // v5: engine has an L2C (set_l2c)
 525   uint64_t l2c_base_ = 0;                     // v5: L2C MMIO register base
 526   hybrid_state_v1 baseline_{};                // LG1: engine reset-state drain
 527   bool has_baseline_ = false;                 // baseline_ valid only when set
 528   bool v_cfg_at_reset_ = true;                // false: capture clobbered vcfg
 529   uint64_t baseline_skip_count_ = 0;          // LG1: elided tolerant writes
 530   uint32_t last_observed_cmderr_ = 0;
 531   uint64_t cmderr_clear_count_   = 0;
 532 
 533 public:
 534   // B17: diagnostic side channel. Non-zero last_observed_cmderr() means
 535   // at least one abstract write tripped a sticky cmderr the restore
 536   // tolerated by clearing. cmderr_clear_count() is the total number of
 537   // such clears across all calls into clear_cmderr_if_set() this run.
 538   auto last_observed_cmderr() const -> uint32_t { return last_observed_cmderr_; }
 539   auto cmderr_clear_count() const -> uint64_t   { return cmderr_clear_count_; }
 540 };
 541 
 542 // Read a 64-bit hart register (GPR / FPR / CSR by abstract regno) over DMI
 543 // via an Abstract Command. Halts-and-reads contract: caller must have the
 544 // hart in the halted state. Shared by StateDrain (drain at exit). If
 545 // abstractcs.cmderr goes sticky after the command, the helper clears it and
 546 // reports 0 rather than blocking the caller -- a missing CSR mirrors as zero
 547 // instead of corrupting the adjacent field.
 548 inline auto read_register_64_checked(DmiBus& bus, uint16_t regno,
 549                                      const char* poll_label
 550                                        = "read_register_64")
 551   -> std::optional<uint64_t>
 552 {
 553   using RD = ResumeDriver;
 554   bus.write(RD::kCommand,
 555             access_register_command(regno, AarSize::Bits64, /*write=*/false));
 556   poll_until_clear(bus, RD::kAbstractcs, RD::kAbstractcsBusy, poll_label);
 557   const auto status = bus.read(RD::kAbstractcs);
 558   if ((status & RD::kAbstractcsCmderrMask) != 0u) {
 559     bus.write(RD::kAbstractcs, RD::kAbstractcsCmderrMask);
 560     return std::nullopt;
 561   }
 562   const uint64_t lo = bus.read(RD::kData0);
 563   const uint64_t hi = bus.read(RD::kData1);
 564   return lo | (hi << 32);
 565 }
 566 
 567 inline auto read_register_64(DmiBus& bus, uint16_t regno,
 568                              const char* poll_label = "read_register_64")
 569   -> uint64_t
 570 {
 571   return read_register_64_checked(bus, regno, poll_label).value_or(0u);
 572 }
