cmake_minimum_required(VERSION 3.20)
project(dispatch CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
option(ENABLE_COVERAGE "Build with gcov instrumentation" OFF)

add_executable(dispatch_test tests/dispatch_test.cpp src/dispatch.cpp)
target_include_directories(dispatch_test PRIVATE src)
target_compile_options(dispatch_test PRIVATE -Wall -Wextra)
if(ENABLE_COVERAGE)
  target_compile_options(dispatch_test PRIVATE --coverage -O0)
  target_link_options(dispatch_test PRIVATE --coverage)
endif()
add_test(NAME dispatch_test COMMAND dispatch_test)
